	title = ww1_berlin_baghdad.2.t
	desc = ww1_berlin_baghdad.2.d
	picture = GFX_report_event_GER_TUR_military_mission
	is_triggered_only = yes
	option = { #Proceed immediately
		name = ww1_berlin_baghdad.2.a
		clr_country_flag = cant_build_berlin_baghdad
		activate_mission = berling_baghdad_construction_mission
		add_ideas = GER_berlin_baghdad_railway_construction

			every_country = {
				limit = {
					has_country_flag = berlin_baghdad_opposes

						has_war_with = GER
					}
				}

					id = ww1_berlin_baghdad.4
					days = 1
				}
			}
		}
	}
}
