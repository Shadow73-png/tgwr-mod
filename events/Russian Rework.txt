				set_temp_variable = { RUS_modify_loyalty_far_right_var = 5 }
				RUS_modify_loyalty_far_right = yes
			}
			if = {
				limit = {
					is_russia_a_monarchy = yes
					NOT = {

					}
				}
				country_event = {
					id = ww1_russia_war.2
					days = 1
				}
			}
		}
	}
	timeout_days = 4

	option = {
		name = ww1_russia_war.1.a
		effect_tooltip = {
			add_timed_idea = {
				idea = RUS_sacred_unity
				days = 365
			}
			if = {
				limit = {
					any_enemy_country = {
						OR = {
							original_tag = GER
							is_in_faction_with = GER
						}	
					}
				}
				set_province_name = {
					id = 3151
					name = PETROGRAD_PROV
				}
			}
			if = {
				limit = {
					NOT = {
						has_idea = war_economy
						has_idea = tot_economic_mobilisation
					}
				}
				add_ideas = partial_economic_mobilisation
			}
			if = {
				limit = {
					duma_is_active = yes

				RUS_modify_loyalty_center_left = yes
