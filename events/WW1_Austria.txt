# Austria Navy is no more
country_event = {
	id = ww1_austria.16
	title = ww1_austria.16.t
	desc = ww1_austria.16.d
	picture = GFX_report_event_destroyers
	fire_only_once = yes
	trigger = {
		tag = AUS
		#lost
		has_war = no
		is_subject = no
		is_great_war_ongoing = no
		has_country_flag = great_war_loser
		#no port
		NOT = {
			has_built = {
				type = naval_base
				value < 1
			}
			any_subject_country = {
				has_built = {
					type = naval_base
					value < 1
				}
			}
		}
		#has navy
		has_navy_size = {
			size > 0
		}
		#reciver is valid
		OR = {
			1198 = {
				owner = {
					has_country_flag = great_war_winner
				}
			}
			any_other_country = {
				is_major = yes
				has_country_flag = great_war_winner
				capital_scope = {
					is_on_continent = europe
				}
				has_built = {
					type = naval_base
					value > 0
				}
			}
		}
	}
	option = {
		name = ww1_austria.16.a
		1198 = {
			owner = {
				if = {
					limit = {
						has_country_flag = great_war_winner
					}
					ROOT = {
						transfer_navy = {
							target = PREV
						}
					}
					hidden_effect = {
						country_event = ww1_austria.17
					}
				}
				else = {
					random_country = {
						limit = {
							is_major = yes
							has_country_flag = great_war_winner
							capital_scope = {
								is_on_continent = europe
							}
							has_built = {
								type = naval_base
								value > 0
							}
						}
						ROOT = {
							transfer_navy = {
								target = PREV
							}
						}
						hidden_effect = {
							country_event = ww1_austria.17
						}
					}
				}
			}
		}
	}
}
