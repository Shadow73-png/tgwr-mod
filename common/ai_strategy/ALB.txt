ALB_agressive_expansion = {
	allowed = { original_tag = ALB }
	enable = {
		tag = ALB
		has_war_with = TUR
		has_country_flag = TUR_revolting_country
		has_completed_focus = ALB_taksim_meeting
		has_completed_focus = ALB_secure_regional_loyalties
	}
	abort_when_not_enabled = yes
	ai_strategy = {
		type = front_control
		state_trigger = { is_core_of = ALB }
		priority = 1000
		ratio = 2
		ordertype = front
		execution_type = rush
		#execute_order = yes
	}
}
