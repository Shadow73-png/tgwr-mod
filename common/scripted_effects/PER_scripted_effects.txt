PER_calculate_total_foreign_influnce = {
	set_variable = { var = PER_total_foreign_influence value = 0 }
	if = {
		limit = { 
			ENG = {
				exists = yes
				NOT = { has_war_with = ROOT }
				has_dynamic_modifier = PER_influence_in_persia_modifier
			}
		}
		add_to_variable = { var = PER_total_foreign_influence value = ENG.PER_foreign_influence }
	}
	if = {
		limit = { 
			NOT = { original_tag = ARB }
			RUS = {
				exists = yes
				NOT = { has_war_with = ROOT }
				has_dynamic_modifier = PER_influence_in_persia_modifier
			}
		}
		add_to_variable = { var = PER_total_foreign_influence value = RUS.PER_foreign_influence }
	}
	if = {
		limit = {
			NOT = { original_tag = ARB }
			GER = {
				exists = yes
				NOT = { has_war_with = ROOT }
				has_dynamic_modifier = PER_influence_in_persia_modifier
			}
		}
		add_to_variable = { var = PER_total_foreign_influence value = GER.PER_foreign_influence }
	}
	if = {
		limit = { 
			TUR = {
				exists = yes
				NOT = { has_war_with = ROOT }
				has_dynamic_modifier = PER_influence_in_persia_modifier
			}
		}
		add_to_variable = { var = PER_total_foreign_influence value = TUR.PER_foreign_influence }
	}
	clamp_variable = {
		var = PER_total_foreign_influence
		min = 0
		max = 1
	}
}
PER_calculate_foreign_influnce_modifiers = {
	set_variable = { var = PER_political_power_gain value = PER_total_foreign_influence }
	multiply_variable = { var = PER_political_power_gain value = -1 }
	#set_variable = { var = PER_consumer_goods_expected_value value = PER_total_foreign_influence }
	#multiply_variable = { var = PER_consumer_goods_expected_value value = 0.1 }
	set_variable = { var = PER_civilian_intel_to_others value = PER_total_foreign_influence }
	multiply_variable = { var = PER_civilian_intel_to_others value = 100 }
	
	
}
PER_calculate_foreign_influnce = {
	PER_calculate_total_foreign_influnce = yes
	PER_calculate_foreign_influnce_modifiers = yes
}

PER_calculate_influnce_in_persia_modifiers = {
	set_variable = { var = PER_political_power_gain value = PER_foreign_influence }
	multiply_variable = { var = PER_political_power_gain value = 0.2 }
	set_variable = { var = PER_consumer_goods_factor value = PER_foreign_influence }
	multiply_variable = { var = PER_consumer_goods_factor value = -0.1 }
}


PER_improve_foreign_advisors = {
	if = {
		limit = { has_idea = PER_foreign_advisors_0 }
		swap_ideas = {
			remove_idea = PER_foreign_advisors_0
			add_idea = PER_foreign_advisors_1
		}
	}
	else_if = {
		limit = { has_idea = PER_foreign_advisors_1 }
		swap_ideas = {
			remove_idea = PER_foreign_advisors_1
			add_idea = PER_foreign_advisors_2
		}
	}

		limit = { has_idea = PER_foreign_advisors_2 }
		swap_ideas = {

	else_if = {
		limit = { has_idea = PER_foreign_advisors_3 }
		swap_ideas = {
			remove_idea = PER_foreign_advisors_3
			add_idea = PER_foreign_advisors_4
		}
	}
	else_if = {
		limit = { has_idea = PER_foreign_advisors_4 }
		swap_ideas = {
			remove_idea = PER_foreign_advisors_4
			add_idea = PER_foreign_advisors_5
		}
	}

		limit = { has_idea = PER_foreign_advisors_5 }

			remove_idea = PER_foreign_advisors_5
			add_idea = PER_foreign_advisors_6

}
