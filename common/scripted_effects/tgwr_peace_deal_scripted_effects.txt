### ENTENTE ###
change_entente_leader = {
	# call from surrendered leader
	if = {
		limit = {
			is_faction_leader = yes
		}
		if = {
			limit = {
				any_other_country = {
					is_in_faction_with = ROOT
					is_subject = no
					has_war_together_with = ROOT
				}
			}
			random_state = {
				prioritize = {
					is_on_same_continent_as = ROOT
					OWNER = {
						OR = {
							tag = FRA
							tag = RUS
							tag = ENG
							tag = ITA
							tag = USA
							is_major = yes
						}
					}
				}
				limit = {
					is_capital = yes
					OWNER = {
						is_in_faction_with = ROOT
						is_subject = no
						has_war_together_with = ROOT
					}
				}
				OWNER = {
					set_faction_leader = yes
					set_country_flag = entente_leader
					every_country = {
						limit = {
							is_exiled_in = ROOT
						}
						end_exile = THIS
						become_exiled_in = { target = PREV }
					}
				}
				ROOT = {
					clr_country_flag = entente_leader
					leave_faction = yes
					every_subject_country = {
						leave_faction = yes
					}
				}
			}
		}
		else = {
			# entente lost
		}
	}
}
