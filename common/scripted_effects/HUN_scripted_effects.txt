HUN_voting_on_nmp_favour_law = {
	if = {
		limit = { 
			has_completed_focus = HUN_national_work_party
			has_global_flag = HUN_compromise_gov_in_charge
		}
		set_temp_variable = {
			HUN_nmp_seats_temp = HUN_total_nmp_seats
		}
		multiply_temp_variable = {
			HUN_total_nmp_seats = 0.9
		}
		add_to_variable = {
				HUN_nmp_seats_temp = 15 #OA and Independent support
			}
		add_to_variable = {
			HUN_parliment_in_favour_voters = HUN_nmp_seats_temp
		}
		round_variable = HUN_parliment_in_favour_voters
		else = {
			set_temp_variable = {
				HUN_nmp_seats_temp = HUN_nmp_seats
			}
			multiply_temp_variable = {
				HUN_nmp_seats_temp = 0.9
			}
			add_to_variable = {
				HUN_nmp_seats_temp = 15 #OA and Independent support
			}
			add_to_variable = {
				HUN_parliment_in_favour_voters = HUN_nmp_seats_temp
			}
			round_variable = HUN_parliment_in_favour_voters
		}
	}
}
