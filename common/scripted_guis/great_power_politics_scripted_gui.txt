scripted_gui = {
	great_power_politics_gui = {
		context_type = decision_category
		window_name = "great_power_politics_gui_window"
		effects = {
			great_power_politics_misc_button_click = {
				if = {
					limit = { has_country_flag = great_power_politics_misc_visible }
					clear_all_great_power_politics_flags = yes
					set_variable = { great_power_politics_misc_button = 1 }
				}
				else = {
					clear_all_great_power_politics_flags = yes
					set_country_flag = great_power_politics_misc_visible
					set_variable = { great_power_politics_misc_button = 2 }
				}
			}
			great_power_politics_great_war_button_click = {
				if = {
					limit = { has_country_flag = great_power_politics_great_war_visible }
					clear_all_great_power_politics_flags = yes
					set_variable = { great_power_politics_misc_button = 1 }
				}
				else = {
					clear_all_great_power_politics_flags = yes
					set_country_flag = great_power_politics_great_war_visible
					set_variable = { great_power_politics_misc_button = 2 }
				}
			}
			great_power_politics_berlin_baghdad_button_click = {
				if = {
					limit = { has_country_flag = great_power_politics_berlin_baghdad_visible }
					clear_all_great_power_politics_flags = yes
					set_variable = { great_power_politics_misc_button = 1 }
				}
				else = {
					clear_all_great_power_politics_flags = yes
					set_country_flag = great_power_politics_berlin_baghdad_visible
					set_variable = { great_power_politics_misc_button = 2 }
				}
			}
		}
		triggers = {
			great_power_politics_misc_button_click_enabled = {	
			}
			great_power_politics_great_war_button_click_enabled = {	
			}
			great_power_politics_berlin_baghdad_button_click_enabled = {		
			}
		}
	}
}
