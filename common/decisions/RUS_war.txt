
TGWR_great_power_politics = {
	#Food shortages
	RUS_food_supply_deteriorates_mission = {
		icon = GFX_decision_generic_consumer_goods
		allowed = { tag = RUS }
		activation = {
			has_country_flag = RUS_food_shortages_mechanic
			has_war_with_major = yes
			has_idea = RUS_inadequate_railway_system
			check_variable = {
				RUS_food_supply > RUS_railway_modernization_progress
			}
		}
		available = {
			OR =  {
				has_war_with_major = no
				NOT = {
					has_idea = RUS_inadequate_railway_system
				}
				check_variable = {
					var = RUS_railway_modernization_progress
					value = RUS_food_supply
					compare = greater_than_or_equals
					tooltip = RUS_food_supply_check
				}
			}
		}
		is_good = no
		days_mission_timeout = 30
		selectable_mission = no
		timeout_effect = {
			if = {
				limit = {
					NOT = {
						has_idea = RUS_food_shortages
					}
				}
				add_ideas = RUS_food_shortages
			}
			set_temp_variable = {
				RUS_add_food_supply_var = -5
			}
			RUS_add_food_supply = yes
		}
	}
	RUS_commandeer_military_trains = {

			has_country_flag = RUS_food_shortages_mechanic
			has_war_with_major = yes
			has_idea = RUS_inadequate_railway_system
		}
		complete_effect = {
			add_equipment_to_stockpile = {

			}
		}
		remove_effect = {
			add_equipment_to_stockpile = {
				type = train_equipment

			}
			set_temp_variable = {
				RUS_add_food_supply_var = 5
			}
			RUS_add_food_supply = yes
		}
	}
