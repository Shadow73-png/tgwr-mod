
#RUSSIA
TGWR_great_power_politics = {
	RUS_occupation_of_nothern_persia_timeout = {
		icon = generic_planted_spear
		allowed = {
			always = no
		}

		activation = {
			always = no
		}

		#targets = { PER }

		highlight_states = {
			highlight_states_trigger = {
				has_contested_owner = PER
				is_owned_by = ROOT			
			}
		}

		available = {
			OR = {
				has_war_with = PER
				has_war_with = TUR
			}			
		}

		cancel_trigger = {
			hidden_trigger = {
				NOT = {
					any_owned_state = {
						has_contested_owner = PER
					}
				}
			}		
		}

		fire_only_once = yes

		is_good = no
		days_mission_timeout = 600

		timeout_effect = {
			every_owned_state = {
				limit = {
					has_contested_owner = PER
				}
				transfer_state_to = PER
			}
		}
	}
	RUS_invasion_of_persia = {
		icon = generic_planted_spear
		allowed = {
			always = no
		}

		activation = {
			always = no
		}

		targets = { PER }

		highlight_states = {
			highlight_state_targets = { state = 266 }
			highlight_provinces = { 10837 }
			highlight_only_provinces = yes
		}

		available = {
			OR = {
				controls_province = 10837
				266 = {
					any_neighbor_state = {
						is_controlled_by = ROOT
					}
				}
			}
		}

		fire_only_once = yes

		is_good = no
		days_mission_timeout = 30

		complete_effect = {
			if = {
				limit = {
					NOT = {
						controls_province = 10837
					}
				}
				PER = {
					country_event = ww1_per_rus_strangling.8
				}
			}
			else = {
				activate_targeted_decision = {
					target = PER
					decision = RUS_secure_tehran
				}
				PER = {
					activate_targeted_decision = {
						target = RUS
						decision = PER_hold_tehran
					}
				}
			}
		}

		timeout_effect = {
			white_peace = PER
			every_owned_state = {
				limit = {
					has_contested_owner = PER
				}
				transfer_state_to = PER
			}
		}
	}
	RUS_secure_tehran = {
		icon = generic_planted_spear
		allowed = {
			always = no
		}

		activation = {
			always = no
		}

		targets = { PER }

		highlight_states = {
			highlight_state_targets = { state = 266 }
			highlight_provinces = { 10837 }
			highlight_only_provinces = yes
		}

		available = {
			hidden_trigger = {
				always = no
			}
			controls_province = 10837
		}

		fire_only_once = yes

		is_good = yes
		days_mission_timeout = 14

		timeout_effect = {
			if = {
				limit = { 
					controls_province = 10837
				}
				PER = {
					country_event = ww1_per_rus_strangling.11
				}
			}
			else = {
				white_peace = PER
				every_owned_state = {
					limit = {
						has_contested_owner = PER
					}
					transfer_state_to = PER
				}
			}
		}
	}
